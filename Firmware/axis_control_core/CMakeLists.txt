cmake_minimum_required(VERSION 3.16)
project(axis_control_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(axis_control_objects OBJECT axis_control_core.cpp identification_model.cpp servo.cpp)
set_target_properties(axis_control_objects PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(axis_control_objects PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(axis_control_objects PRIVATE -Wall -Wextra -Werror)
add_library(axis_control_core STATIC $<TARGET_OBJECTS:axis_control_objects> replay.cpp)
target_include_directories(axis_control_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(axis_control_core_native SHARED $<TARGET_OBJECTS:axis_control_objects>)
set_target_properties(axis_control_core_native PROPERTIES OUTPUT_NAME axis_control_core)
add_executable(commissiond commissiond.cpp)
target_link_libraries(commissiond PRIVATE axis_control_core)
add_executable(test-servo tests/test_servo.cpp)
target_link_libraries(test-servo PRIVATE axis_control_core)
add_test(NAME servo_core COMMAND test-servo)
