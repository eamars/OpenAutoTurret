cmake_minimum_required(VERSION 3.16)
project(axis_control_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Pure servo core (no yaml): the yaw position servo and the pitch position loop.
add_library(axis_control_core STATIC servo.cpp position_loop.cpp)
target_include_directories(axis_control_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(axis_control_core PRIVATE -Wall -Wextra -Werror)
add_executable(commissiond commissiond.cpp)
target_link_libraries(commissiond PRIVATE axis_control_core)
# Servo asset loader, plant models and the session simulator (yaml-cpp).
add_library(servo_tools STATIC servo_config.cpp plant.cpp simulate.cpp)
target_include_directories(servo_tools PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(servo_tools PUBLIC axis_control_core yaml-cpp)
target_compile_options(servo_tools PRIVATE -Wall -Wextra -Werror)
add_executable(servo-sim servo_sim.cpp)
target_link_libraries(servo-sim PRIVATE servo_tools)
add_executable(test-servo tests/test_servo.cpp)
target_link_libraries(test-servo PRIVATE axis_control_core)
add_test(NAME servo_core COMMAND test-servo)
add_executable(test-plant tests/test_plant.cpp)
target_link_libraries(test-plant PRIVATE servo_tools)
add_test(NAME servo_plant_model COMMAND test-plant)
